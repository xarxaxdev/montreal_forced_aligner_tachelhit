ⴽⵓ ⵎⴰ ⴰⴷ ⵉⴷⵔⵓⵙⵏ ⵓⵔ ⵉⵜⵜⵓⵇⴱⵉⵍ.
ⴰⴷ ⵏⵎⵢⴰⵏⵏⴰⵢ ⴳ ⵓⵍⴰⵡⴰⵢ.
ⵓⵔ ⴰⴷ ⴷⴳⴳⵉⵏ, ⴰⵛⴽⵓ ⵢⴰⴳⴳⵓⴳ ⴱⴰⵀⵔⴰ.
ⵎⴰⵏⵉ ⴳ ⵜⵍⵍⴰⵎ?
ⵓⵔ ⴷⴰ ⵏⴽⵜⵜⵉ ⵉⵣⵍⴰⵏ ⴰⵔ ⴷ ⵉⵙⵎⵓⵔ ⵓⵃⵉⴷⵓⵙ
ⴷⴷⴰⵏ ⵉⴹⴳⴰⵎ.
ⴷⴰⵔⵉ ⵉⵎⴷⴷⵓⴽⴽⴰⵍ ⴳ ⵎⵏⵛⴽ ⵏ ⵜⵎⵓⵔⴰ.
ⵉⵙⴰⵇⵙⴰ ⵜⵓⵎ ⵅⴼ ⵎⴰⵏ ⴰⵢⴰ.
ⵜⴼⵊⵊⵉⵊⴷ?
ⵓⵙⵉⵏ ⵉⵡⴹⴼⴰⵏ ⵜⵉⵔⵛⵜ ⵏ ⵉⵎⵏⴷⵉ
ⴽⵓ ⴰⵙⵉⵎⵡⴰⵙ ⵍⴰ ⵏⵜⵜⴻⵜⵜ ⵙⴽⵙⵓ
ⵉⵣ ⴷ ⵜⴰⵡⵊⴰ ⵏⵏⴽ ⴰⵢⴰ?
ⵔⴰⴷ ⴽⵯⵏ ⵏⵙⵙⵖⵜⵎ ⵙ ⵜⵡⵍⴰⴼⵜ.
ⴽⵔⴰ ⵏ ⵜⵉⴽⴽⴰⵍ ⴰⵔ ⵉⵜⵜⴰⵔⴰ ⵜⴰⴱⵔⴰⵜ ⵉ ⵎⴰⵙ.
ⴰ ⴰⴽⵓⵛ ⵉⵏⵓ!
ⵜⴰⵎⵙⵓⵍⴰⴼⵜ ⵏⵏⴰ ⵜⵙⵖⵉⴷ ⵜⵓⴼ ⵜⵉⵏⵓ.
ⵏⴰⵔⴰ ⴰⵎⴰⴳⵔⴰⴷ ⵏⵏⵖ ⵉⴹⴳⴰⵎ.
ⴽⵓⵏⵏⵉ ⴷ ⵉⵔⵓⵙⵉⵢⵏ?
ⵔⵉⵖ ⴰⵏⵙⵎⴰⴷⴷⵓ ⵏ ⵓⴽⴼⴰⵢ.
ⵙⵓⵍ ⵓⴽⴰⵏ ⵉⴼⵜⵜⵓ ⵜⵓⵎ.
ⵢⵓⴳⵔ ⵃⵎⵎⵉ ⴰⵥⵔⵡⴰⵍ.
ⵉⴳⴰ ⵟⵟⴰⵍⴱ.
ⴳⵉⵖ ⵢⴰⵏ ⵓⵎⴷⴰⵏ ⴰⵎⵅⵢⵢⵏ.
ⵣⵎⵔⵖ ⴰⴷ ⴰⵡⵏ ⴰⵣⵏⵖ ⴰⵎⴰⵡⴰⵍ ⵉⵏⵓ ⵏ ⵜⵎⴰⵣⵉⵖⵜ.
ⵍⴰ ⵜⴻⵜⵜⵕⴰⵊⵓ ⴰⴷ ⵜⵕⴰⵃ ⴰⴽⴷ ⵜⵙⵍⵉⵜ.
ⴰⴽⴰⴷ ⵉⵎⵎⵉⵎ ⵓⵜⵜⵛⵉ.
ⵢⵉⵡⵔⵔⵉ ⴷ ⵙ ⵜⵥⵓⵕⵉ.
ⵖⵔⵉⵖ ⵉⴹⵍⵍⵉ ⵉⵛⵜ ⵏ ⵜⵓⵍⵍⵉⵙⵜ ⵉⵙⵜⴰⵡⵀⵎⵎⴰⵏ.
ⵏⵛⵛ ⵖⴰⵙ ⴷ ⴰⵎⴰⵍⵍⴰⵢ.
ⴰⵔ ⵜⵜⵍⵍⵉⵖ ⵙ ⵢⴰⵜ ⵜⴳⵓⵔⵉ ⴳ ⵎⵙⵎⵓⵏ ⴰⵡⴰⵍ.
ⵉⵙⵙⵢⴰⴼⴰ ⴰⴳⵥⵥⴰⵥ ⴳ ⵜⴰⴷⴰⵡⵜ.
ⵎⵉⵔⵉⴽⴰⵏ
ⵜⴰⴹⴰⵔⵜ ⵏ ⵓⴳⵓⵊⵉⵍ ⴰⵢ ⵉⴱⴱⵉⵏ ⴰⴳⵔⵜⵉⵍ
ⴰⴷ ⵓⴽⴰⵏ ⵙⵍⵍⴰⵖ ⵉ ⵜⵣⵍⵉⵜ ⴰⴷ ⴰⵔ ⴷ ⵉⵍⵎⵎⴰ ⴽⵯⵜⵜⵉⵖ ⵜⵉⵎⵥⵉ ⵉⵏⵓ.
ⵓⵙⵔⵖ ⴽⵯⵏ.
ⵜⵓⴽⴽⵕⴹⴰ ⵉⵣⴳⵍⵏ.
ⵜⵉⵢⵜⵉ ⵏ ⵡⵓⵍ ⵉⵊⵀⴷⵏ.
ⴰⵙⴳⴳⵯⴰⵙ ⴰⴷ ⵉⵎⵓⵔⵙ ⵓⴽⴰⵏ ⵉⵡⵉ ⴳ ⵍⴱⴰⴽ.
ⵜⴰⵎⵙⴰⴷⵓⵔⵜ.
ⵉⵍⵍⴰ ⵢⴰⴷⵍⵍⵉ ⵖⵉⴼⴽ ⴰⴷ ⵏⵏ ⵜⵉⵍⵉⴷ ⴳ ⵢⵉⵡⵉⵣ ⵉⴹⵍⵍⵉ. ⴳⵉⵙ ⵜⵓⴳⵜ ⵏ ⵜⵖⴰⵡⵙⵉⵡⵉⵏ ⴼⵊⵊⵉⵊⵏⵉⵏ.
ⵉⵙⵙⵏ ⴰⴷ ⵉⵙⴰⵡⴰⵍ.
ⵉⴷⴷⴰ ⴷⴰ ⵉⴱⵟⵟⵓ ⴰⵍⵉⵎ ⵉ ⵉⵖⵢⴰⵍ
ⵜⵔⴰ ⴰⴷ ⴷⵉⴷⵙⵏ ⵜⵉⵍⵉ.
ⴳⴳ ⵎⵉⵏ ⵜⵅⵙⴷ.
ⵉⵏⵎⵓⵇⵇⴰⵔ ⵉ ⵜⵢⴰⴼⵓⵜⵉⵏ ⵜⵉⵏⴼⵍⴰⵃⵉⵏ.
ⵢⴰⵏ ⵉⵛⵡⴰⵏ ⴰⵔ ⵉⵜⵜⵏⵡⵡⴰ ⵍⵅⵉⵔ ⴳ ⵡⵓⵙⵙⴰⵏ
ⵉⵙ ⵜⴳⴰ ⵜⵡⵓⵔⵉ ⵏⵏⴽ ⵜⵉⵎⵉⵣⵉⵜ?
ⵜⵥⵔⴰ ⵉⵙⵍⵎⴰⵏ, ⵜⴻⵜⵜⵓ ⵙⵍⵉⵎⴰⵏ
ⵓⵔ ⴷⴰ ⵉⵣⴷⴷⵖ ⵡⴰⴽⴰ ⴳ ⵓⵎⴰⵣⵉⵔ ⵏ ⵉⴼⵉⵖⵔ
ⴰⵙⵉⵢⴰⵜ ⵍⵖⴰⵛⵉ ⴰⵏⵏ ⵙ ⵜⴳⵍⴰⵍⴰⵣⵜ.
ⴰⴳⴰ ⵉⴳⵯⴱⴰⵏ ⵓⵔ ⴷⴰ ⵉⵜⵜⴰⴳⵎ
ⴰⵎⵎⴰⵙ ⵏ ⵓⴹⴼⴼⵓⵕ ⴷ ⵓⵙⴼⵙⵉ ⵏ ⵉⵙⵓⵜⴰⵔ.
ⵎⴰⵏⵉ ⵙ ⵔⴰⴷ ⵜⴷⴷⵓⴷ?
ⴰⵙⵍⵍⵎ
ⵜⵏⵊⵎⴻⵎⵜ ⴰⴷ ⵉⵢⵉ ⴷ ⵜⵙⵙⵓⵖⵍⵎⵜ ⴽⵔⴰ ⵏ ⵜⵡⵉⵏⴰⵙ?
ⵎⴰ ⵜⵔⵉⴷ.
ⵉⵣⴳⴰⵔⵏ ⵏ ⴷⴰⴷⵙ ⴰⵢ ⵉⴽⵔⵔⵣⵏ ⴰⴽⴰⵍ ⵏ ⴷⴰⴷⵙ
ⴰⵔ ⵏⵙⵡⵉⵏⴳⵉⵎ ⵏⴽⴽ ⴷ ⵎⴰⵔⵉ ⴰⴷ ⵏⵙⵙⴳⵎⵉ ⵙⵉⵏ ⵡⴰⵣⵣⴰⵏⵏ.
ⵜⴰⵎⴰⵡⴰⵙⵜ ⵏ ⵓⵡⴰⵏⴽ.
ⴳⴳⵣⴰⵜ, ⵜⵙⵙⴼⵍⴷⵎ ⵉ ⵓⵙⵍⵎⴰⴷ.
ⴰⵍⵍⵏ ⵏ ⵓⵏⴱⴳⵉ ⴰⴷ ⵉⵙⵙⵏⵡⴰⵏ ⵉⵎⵏⵙⵉ
ⴷⴷⵓ ⵙ ⵜⵜⴰⵡⵉⵍ
ⵓⵔ ⵙⵍⵍⵉⵖ ⵉ ⵎⵉⵏ ⵜⵏⵏⵉⴷ.
ⴷⴰⴷ ⵏⵙⵖⴰⵡⵙⴰ ⵙ ⵜⵡⵉⵣⵉ ⵏⵏⴽ.
ⵙⵍⵍⴰⵖ ⵉ ⴽⵔⴰ ⴰⵔ ⵉⵜⵜⵓⵙⵓ.
ⵓⵔ ⵖⵓⵔⵙⵏⵜ ⵉⵙⵍⴰⵃⵏ, ⵎⴰⵛⴰ ⴷⴰ ⵙⴽⵏⵓⵏⵓⵢⵏⵜ ⵉⵥⵔⴰⵏ, ⴰⵔ ⵙⵃⵓⵔⵔⵓⵛⵏⵜ ⵉⵔⴳⵣⵏ ⵙ ⵉⵣⵍⵉ ⴷ ⵜⵖⵔⴰⵜⵉⵏ.
ⵎⴰⵏⵎⴽ ⵙ ⵔⴰⴷ ⴱⴷⵔⵖ?
ⴽⵓⴷ ⵏⵜⵜⵉⴷⵉⵔ, ⵏⵜⵜⵉⵥⵉⵔ
ⵜⴰⴳⴳⵓⴳ ⴷⴷⵣⴰⵢⵔ ⴼ ⵍⵀⵏⴷ.
ⴷⴰⵖ.
ⵏⵏⵓⵔⵣⵖ ⴰⴷ ⵜ ⴰⴽⴽⵯ ⵓⵔ ⵜⵏⵏⵉⵜ ⵉ ⵜⵓⵎ ⵎⴰⵢⴰⵏⵏ.
ⵎⵉⵏ ⵜⵙⵙⵉⵔⵉⵎⴷ?
ⵉⵙⵡⵉⵜⵏ ⵉⵏⵛⵏⴳⴰⵏⵏ ⴰ ⴼ ⵜⵜⵡⴰⵙⴽⴰⵔⵏⵜ ⵜⵙⵉⵍⴰ ⵜⵉⵙⴰⵔⴷⴰⵙⵉⵏ ⵜⵉⵎⵉⵔⵉⴽⴰⵏⵉⵢⵉⵏ ⴳ ⴽⵔⴰⵢⴳⴰⵜ ⵜⴰⵙⴳⴰ ⴳ ⵓⵎⴰⴹⴰⵍ.
ⵉⵙ ⴷ ⵍⵀⵏⵓⴷ ⴰⴷ ⵉⵙⴽⵔⵏ ⴳⴰⵕⵕⵓ?
ⵏⵏⵔⴰⵏⵜⵏ ⵏⵉⵜ ⴷⴰⵖ.
ⵓⵔ ⵡⴰⵜⵙⵖ ⴰⴷ ⴽ ⴱⵔⵉⵖ.
ⵓⵔ ⵉⵍⵍⵉ ⵓⵎⵍⵍⴰⵍ ⴳ ⵉⴳⵓⵔⴷⴰⵏ
ⴰⴷ ⵏⵙⵎⵓⵏ ⴰⵔⴷⵍⵉⵎ ⵏⵏⵉ.
ⵜⵉⵖⵣⵉ ⵏ ⵣⵣⵎⴰⵏ ⴰⴷ ⴽ ⵉⴷ ⵉⵙⵙⴽⵯⵏⴰⵏ ⴰ ⴰⵖⴰⵏⵉⵎ.
ⵜⴰⵏⵎⵎⵉⵔⵜ.
ⵓⵔ ⴷⴰ ⵜⵜⴷⵔⴰⵡⵏ ⵉⵢⴹⴰⵏ
ⵉⴽⴽⴰ ⵜⵜ ⵉⵏⵏ ⵙⴰⵎⵉ ⵉⵔⴰ ⴰⴷ ⴰⴽⴽⵯ ⵢⴰⵔⵎ ⴽⵔⴰⵢⴳⴰⵜ.
ⵣⴳⴳⵉⵖ ⵎⴰⵙ ⵔⴰⴷ ⵉⵣⵣⵔⵉ ⵢⵉⵔⵉⵎ.
ⵉⵏⵉ ⴰⵖ ⵓⴳⴳⴰⵔ.
ⵓⵔ ⵉⴳⵉ ⵓⵍⵎⵓⵇⵇⴰⵔ ⴰⴷ ⵎⴰⴷ ⵏⵜⵜⴼⵍⵍⴰⵜ.
ⵉⵍⵍⴰ ⵖⵉⴼⵏⵖ ⴰⴷ ⵏⵙⴱⵉⴷⴷ ⵎⴰⵢⴰⴷ ⴷⴰⵜ ⴰⴷ ⵉⵎⵙⴰⵔ.
ⴰⴷ ⵓⵔ ⵜⵙⵎⵓⵎⵎⵓⵢⴷ.
ⵜⵙⴽⵔ ⵉⵢⵉ ⵙⵜⵜⵉ ⵢⴰⵏ ⵓⴱⴱⵓⵏⵉⵣ ⵉⵥⵡⵉⵥⵥⵉⵢⵏ.
ⵜⴰⵎⵀⵍⴰ ⵏ ⵜⵖⵓⵔⵉⵡⵉⵏ ⴷ ⵓⵎⵢⴰⵡⴰⵙ ⴷ ⵓⵙⵜⵔⴰⵔ.
ⵉⵙ ⵜⵎⵎⵔⴷ?
ⴰⵙⵍⵎⴰⴷ ⵏ ⵜⵓⴱⵉⵔⵜ ⴰⵢ ⵉⴳⴰ.
ⴰⴷ ⵓⵔ ⵜⵓⵚⵚⴰⵜ ⴰⴳⵓⵊⵉⵍ ⵅⴼ ⵉⵎⵟⵟⴰⵡⵏ
ⵜⵓⵎ ⴰⵢⴷ ⵉⴳⴰⵏ ⴰⵏⵎⵀⴰⵍ ⵏ ⵜⵔⴱⵉⵄⵜ.
ⴰⵎⵙⴰⴼⴰⴹ, ⴰⴷ ⴰⴽⴽⵯ ⵉⵅⵛⵏⵏ ⵉⵛⵇⵇⵓⵏ ⴳ ⵜⵓⴷⵔⵜ.
ⵉⴽⴽⴰ ⵜⵜ ⵉⵏⵏ ⵎⴰⵏⵏⴰⴷ ⴳ ⵢⴰⵏ ⵡⴰⴷⴷⴰⴷ ⴰⵀⵉⵙⵜⵉⵔⵉ.
ⵍⴰ ⵥⵕⵕⴰⵖ ⴰⵎⵃⴹⴰⵕ ⵉⵃⵕⵛⵏ.
ⵀⴰ ⵉⵢⵉ ⴷⴷⵉⵖ ⵙ ⵢⴰⵏ ⵓⵎⵎⵓⴷⴷⵓ.
ⵀⴰⵜ ⵏⵜⵜⴰⵜ ⴰⵢⴷ ⵉⵙⵔⵙⵏ ⵜⴰⵙⵖⵓⵏⵜ ⵅⴼ ⵜⴷⴰⴱⵓⵜ.
ⵜⵓⴷⵔⵜ ⵏⵏⵖ ⴰ ⵖ ⵏⵍⵍⴰ.
ⵇⵇⴰⴷ ⴳⴳⵖ ⵎⵉⵏ ⵜⵅⵙⴷ.
ⵉⵍⴰ ⴰⵏⴰⵙⵉⵡ ⵏ ⵜⵎⴰⵣⵉⵖⵜ ⴳ ⵓⵎⵙⵙⵓⴷⵙ ⵏⵏⵙ.
ⵖⵍⵙⵖ ⵉ ⴽⵢⵢ?
ⴰⵎⵛⵛⵉⵡⵕ.
ⵉⴽⴽⴰ ⵜⵜ ⵉⵏⵏ ⵜⵓⵎ ⵜⵣⵡⴰⵔ ⴰⵙ ⴱⴷⴷⴰ ⵜⴰⵡⵏⵣⴰ ⵎⴰ ⵜⵍⴰ ⵜⵓⴷⵔⵜ ⵏⵏⵙ.
ⵓⵔ ⵉⴼⵓⵍⴽⵉⵏ ⵢⴰⵏ ⵉⵏⵇⵇⴰⵏ ⵜⵉⴽⵟⵟⵓⴼⵉⵏ ⴰ ⵜⴰⵔⵡⴰ.
ⵓⵔ ⵉⵙⵙⵉⵏ ⵎⴰ ⴳ ⵉⴽⵛⵎ ⵉⴼⵉⵍⵓ ⵜⴰⴱⴰⵔⴷⴰ
ⵓⵔ ⴷⴰ ⵉⵜⵜⵓⵖⵓⵍ ⴰⵎⴰⵏ ⵖⴰⵙ ⴰⵖⵢⵓⵍ
ⵢⵓⵛⴰ ⴰⵖ ⴰⵎⵛⵉⵛ ⴰ.
ⵉⵙⴰⴼⴼⵏ ⵏ ⴷⵔⴰ, ⴷⴰⴷⵙ, ⵣⵉⵣ,ⵖⵔⵉⵙ ⴷ ⴳⵉⵔ.
ⵓⵔ ⵜⵙⵡⵉⵜ ⵜⴰⵥⴰⵍⵉⵎⵜ.
ⴷⴰ ⴰⴱⴷⴰ ⵉⵜⵜⵉⵍⵉ.
ⵍⵍⴰⵏ ⵓⴽⴰⵏ ⵙⵓⵍ ⵎⵉⴷⴷⵏ ⴳ ⵜⵎⴰⵣⵉⵔⵜ ⴰⴷ ⴰⴷ ⵣⵣⵉⴳⵉⵣⵏ ⵙ ⵉⴱⴰⵛⵉⵍⵏ.
ⵉⵖ ⵜⵣⵎⵔⴷ, ⵜⵔⵡⴰⵃⵜ ⴰⴽⵉⴷⵏⵖ.
ⵙⵡⵉⵏⴳⵎ ⵖ ⴰⵢⵍⵍⵉ ⴰⴽ ⵏⵏⵉⵖ.
ⵓⵔ ⴷⵊⵉⵏ ⵉⵍⵉ ⵉⵙⴽⴰⵡⵏ
ⵓⵔ ⴰⵖ ⵙⵓⵍ ⵉⵜⵜⵖⵏⵏⴰⵊ ⵜⵓⵎ.
ⵉⵍⵓⵥ ⴱⴷⴷⴰ ⵜⵓⵎ ⵏⵜⵜⴰ.
ⴷ ⴰⵙⵍⵎⴰⴷ ⵏ ⵜⵓⴱⵉⵔⵜ.
ⴰⵡⵙ ⴰⵙⵏ
ⵉⴽⵯⵔⵎ ⴱⴰⵀⵔⴰ ⵓⵃⴰⵏⵓ ⵉⵏⵓ.
ⵓⵔ ⵉⵔⵉ ⵙⴰⵎⵉ ⴰⴷ ⵉⵙⵙⵅⵙⴰⵔ ⵉⴷⵔⵉⵎⵏ ⵏⵏⵙ ⵅⴼ ⵉⵢⵙⴰⵏ.
ⵎⵙⵍⵖ ⵉⵖⵔⴱⴰⵏ ⵙ ⵓⵎⵍⵉⵍ.
ⴰⵔ ⵉⵊⴷⴷⵔ ⵓⴽⵛⵛⵓⴹ.
ⵓⴽⵣⵖ, ⵎⴰⵛⴰ ⵓⵔ ⵣⵎⵉⵔⵖ ⴰⴷ ⴷⴳⴳⵉⵖ.
